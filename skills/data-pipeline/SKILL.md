# Data Pipeline Generator

Generate a standalone `pipeline.py` + `pipeline_config.yaml` that automates your data preparation workflow: fetch → validate → transform → output.

Usage: `/data-pipeline [--source <url_or_path>] [--output <dir>]`

Examples:
- `/data-pipeline` (codify the current session's fetch/quality/transform steps)
- `/data-pipeline --source "World Bank GDP data" --output data/`

## Step 1: Gather pipeline requirements

Before generating code, identify:

1. **Data source**: What was fetched? (API URL, database query, file path, public dataset)
2. **Quality checks**: What validation rules apply? (max missing %, required columns, value ranges)
3. **Transformations**: What encodings were applied? (categorical mappings, cyclic conversions, datetime decomposition)
4. **Output format**: Arrow IPC (preferred), CSV, or Parquet
5. **Column types**: The final `column_types` list for jaxcross

If `/data-fetch`, `/data-quality`, and `/data-transform` were run in this session, extract the exact steps from the conversation history.

## Step 2: Generate pipeline.py

Create a Python script with this structure:

```python
#!/usr/bin/env python3
"""Data pipeline: fetch, validate, and transform data for jaxcross modeling.

Generated by /data-pipeline skill.
Re-run with: python pipeline.py --config pipeline_config.yaml
"""

import argparse
import hashlib
import json
import logging
import os
import sys
import time
from datetime import datetime
from pathlib import Path

import numpy as np
import pandas as pd

logging.basicConfig(
    level=logging.INFO,
    format="%(asctime)s [%(levelname)s] %(message)s",
    handlers=[
        logging.StreamHandler(),
        logging.FileHandler("pipeline.log"),
    ],
)
logger = logging.getLogger(__name__)


def load_config(config_path: str) -> dict:
    """Load pipeline configuration from YAML."""
    import yaml
    with open(config_path) as f:
        return yaml.safe_load(f)


# ── Stage 1: Fetch ──────────────────────────────────────────────
def fetch_data(config: dict) -> pd.DataFrame:
    """Download or query data from the configured source."""
    source = config["source"]
    logger.info(f"Fetching data from: {source['type']}")
    
    # -- Implement based on source type --
    # API: requests.get with pagination
    # Database: pd.read_sql with connection
    # File URL: pd.read_csv / pd.read_parquet
    # Local file: pd.read_csv / pd.read_parquet
    
    # REPLACE this with the actual fetch logic from your /data-fetch session
    if source["type"] == "csv":
        df = pd.read_csv(source["path"])
    elif source["type"] == "url":
        df = pd.read_csv(source["url"])
    elif source["type"] == "api":
        # Implement API fetch with pagination
        raise NotImplementedError("Replace with your API fetch logic")
    elif source["type"] == "database":
        # Implement database query
        raise NotImplementedError("Replace with your database query logic")
    else:
        raise ValueError(f"Unknown source type: {source['type']}")
    
    logger.info(f"Fetched {len(df)} rows x {len(df.columns)} columns")
    return df


# ── Stage 2: Validate ──────────────────────────────────────────
def validate_data(df: pd.DataFrame, config: dict) -> pd.DataFrame:
    """Run quality checks. Fail on critical issues, warn on others."""
    rules = config.get("validation", {})
    errors = []
    warnings = []
    
    # Required columns
    required = rules.get("required_columns", [])
    missing_cols = [c for c in required if c not in df.columns]
    if missing_cols:
        errors.append(f"Missing required columns: {missing_cols}")
    
    # Row count minimum
    min_rows = rules.get("min_rows", 10)
    if len(df) < min_rows:
        errors.append(f"Too few rows: {len(df)} < {min_rows}")
    
    # Max missing rate per column
    max_missing = rules.get("max_missing_rate", 0.5)
    for col in df.columns:
        rate = df[col].isna().mean()
        if rate > max_missing:
            warnings.append(f"Column '{col}' has {rate:.0%} missing (>{max_missing:.0%})")
    
    # Duplicate rows
    n_dupes = df.duplicated().sum()
    if n_dupes > 0:
        warnings.append(f"{n_dupes} duplicate rows found")
    
    # Value range checks
    for check in rules.get("value_ranges", []):
        col, vmin, vmax = check["column"], check.get("min"), check.get("max")
        if col in df.columns:
            vals = df[col].dropna()
            if vmin is not None and vals.min() < vmin:
                warnings.append(f"'{col}' has values below {vmin}")
            if vmax is not None and vals.max() > vmax:
                warnings.append(f"'{col}' has values above {vmax}")
    
    # Report
    for w in warnings:
        logger.warning(w)
    if errors:
        for e in errors:
            logger.error(e)
        raise ValueError(f"Validation failed: {errors}")
    
    logger.info(f"Validation passed ({len(warnings)} warnings)")
    return df


# ── Stage 3: Transform ─────────────────────────────────────────
def transform_data(df: pd.DataFrame, config: dict) -> tuple:
    """Apply encodings and produce jaxcross-ready output."""
    transforms = config.get("transforms", {})
    encodings = {}
    
    # Drop columns
    drop_cols = transforms.get("drop_columns", [])
    df = df.drop(columns=[c for c in drop_cols if c in df.columns])
    
    # Categorical encoding
    for col, mapping in transforms.get("categorical_encodings", {}).items():
        if col in df.columns:
            encodings[col] = mapping
            df[col] = df[col].map(mapping)
            logger.info(f"Categorical-encoded '{col}': {len(mapping)} categories")
    
    # Binary encoding
    for col, mapping in transforms.get("binary_encodings", {}).items():
        if col in df.columns:
            encodings[col] = mapping
            df[col] = df[col].map(mapping)
    
    # Ordinal encoding
    for col, mapping in transforms.get("ordinal_encodings", {}).items():
        if col in df.columns:
            encodings[col] = mapping
            df[col] = df[col].map(mapping)
    
    # Cyclic encoding
    for item in transforms.get("cyclic_encodings", []):
        col, period, offset = item["column"], item["period"], item.get("offset", 0)
        if col in df.columns:
            df[col] = ((df[col] - offset) / period) * 2 * np.pi
            logger.info(f"Cyclic-encoded '{col}': period={period}")
    
    # Datetime decomposition
    for col in transforms.get("datetime_columns", []):
        if col in df.columns:
            dt = pd.to_datetime(df[col])
            df[f"{col}_hour"] = dt.dt.hour * 2 * np.pi / 24
            df[f"{col}_dow"] = dt.dt.dayofweek * 2 * np.pi / 7
            df[f"{col}_month"] = (dt.dt.month - 1) * 2 * np.pi / 12
            df[f"{col}_year"] = dt.dt.year.astype(float)
            df = df.drop(columns=[col])
            logger.info(f"Decomposed datetime '{col}'")
    
    # Convert to float32
    for col in df.columns:
        df[col] = pd.to_numeric(df[col], errors="coerce").astype(np.float32)
    
    # Replace infinities with NaN
    df = df.replace([np.inf, -np.inf], np.nan)
    
    # Build column_types list
    column_types = []
    type_map = transforms.get("column_types", {})
    for col in df.columns:
        ct = type_map.get(col, "CONTINUOUS")
        column_types.append(ct)
    
    logger.info(f"Transformed: {len(df)} rows x {len(df.columns)} columns")
    return df, column_types, encodings


# ── Stage 4: Output ────────────────────────────────────────────
def save_output(df: pd.DataFrame, column_types: list, encodings: dict,
                config: dict, output_dir: str):
    """Save transformed data and metadata."""
    os.makedirs(output_dir, exist_ok=True)
    
    output_format = config.get("output", {}).get("format", "arrow")
    base_name = config.get("output", {}).get("name", "prepared")
    
    # Save data
    if output_format == "arrow":
        import pyarrow as pa
        import pyarrow.ipc as ipc
        table = pa.Table.from_pandas(df)
        path = os.path.join(output_dir, f"{base_name}.arrow")
        with pa.OSFile(path, "wb") as f:
            writer = ipc.new_file(f, table.schema)
            writer.write_table(table)
            writer.close()
    elif output_format == "parquet":
        path = os.path.join(output_dir, f"{base_name}.parquet")
        df.to_parquet(path, index=False)
    else:
        path = os.path.join(output_dir, f"{base_name}.csv")
        df.to_csv(path, index=False)
    
    logger.info(f"Data saved to: {path}")
    
    # Save metadata
    meta = {
        "columns": list(df.columns),
        "column_types": column_types,
        "shape": list(df.shape),
        "generated_at": datetime.now().isoformat(),
        "config_hash": hashlib.md5(
            json.dumps(config, sort_keys=True).encode()
        ).hexdigest(),
    }
    meta_path = os.path.join(output_dir, f"{base_name}_metadata.json")
    with open(meta_path, "w") as f:
        json.dump(meta, f, indent=2)
    
    # Save encodings
    enc_path = os.path.join(output_dir, "encodings.json")
    with open(enc_path, "w") as f:
        json.dump(encodings, f, indent=2)
    
    logger.info(f"Metadata saved to: {meta_path}")
    logger.info(f"Encodings saved to: {enc_path}")


# ── Caching ─────────────────────────────────────────────────────
def check_cache(config: dict, output_dir: str) -> bool:
    """Check if output is already up-to-date."""
    meta_path = os.path.join(output_dir, "prepared_metadata.json")
    if not os.path.exists(meta_path):
        return False
    
    with open(meta_path) as f:
        meta = json.load(f)
    
    config_hash = hashlib.md5(
        json.dumps(config, sort_keys=True).encode()
    ).hexdigest()
    
    if meta.get("config_hash") == config_hash:
        logger.info("Output is up-to-date (config unchanged). Use --force to re-run.")
        return True
    return False


# ── Main ────────────────────────────────────────────────────────
def main():
    parser = argparse.ArgumentParser(description="Data pipeline for jaxcross")
    parser.add_argument("--config", default="pipeline_config.yaml", help="Config file path")
    parser.add_argument("--output", default="data/", help="Output directory")
    parser.add_argument("--force", action="store_true", help="Force re-run even if cached")
    args = parser.parse_args()
    
    config = load_config(args.config)
    
    if not args.force and check_cache(config, args.output):
        return
    
    start = time.time()
    logger.info("=" * 60)
    logger.info("Pipeline started")
    
    try:
        df = fetch_data(config)
        df = validate_data(df, config)
        df, column_types, encodings = transform_data(df, config)
        save_output(df, column_types, encodings, config, args.output)
        
        elapsed = time.time() - start
        logger.info(f"Pipeline completed in {elapsed:.1f}s")
        logger.info("=" * 60)
    except Exception as e:
        logger.error(f"Pipeline failed: {e}")
        raise


if __name__ == "__main__":
    main()
```

    - column_a
    - column_b

transforms:
  drop_columns:
    - id
    - timestamp_raw
  
  categorical_encodings:
    color: {"red": 0, "blue": 1, "green": 2}

    - created_at
  
  column_types:
    # Override auto-detection per column

    is_active: BINARY
    education: ORDINAL
    hour: CYCLIC

  format: arrow               # arrow, parquet, csv

## Common Pitfalls

- **Missing `pyyaml`**: Install with `pip install pyyaml` before running the pipeline.
- **API key in config**: Use environment variables, not hardcoded keys. Reference via `os.environ["API_KEY"]` in the fetch function.
- **Schema drift**: If the source schema changes (new columns, renamed columns), the pipeline will fail at validation. Update `required_columns` in config.
- **Timezone-dependent datetime**: Decompose in a consistent timezone (UTC recommended).
